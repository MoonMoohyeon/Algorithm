<< 레벨 풀이

같은 레벨이면 bfs
다른 레벨이면 dfs.

<< 이를 좀더 쉽게 하기 위해서 arr[level]이란 배열을 만들어서 해결
<< level이 다운될 때 arr[level-1] += stk.push() * arr[level];


#include<iostream>
#include<cstring>
#include<stack>
using namespace std;

stack<int> stk;
string str;
bool check = 1;
int arr[31] = {0,};

int main(){
	cin >> str;
	int s = str.size();
	int level = 0;
	
	for(int i=0; i<s; i++){
		if(str[i] == '('){
			stk.push(2);
			if(arr[level] == 1)
				arr[level] = 0;
			level++;
			arr[level] = 1;
		}else if(str[i] == '['){
			stk.push(3);
			if(arr[level] == 1)
				arr[level] = 0;
			level++;
			arr[level] = 1;
		}else if(str[i] == ')'){
			if(stk.empty()){
				check = 0;
				break;
			}
			
			int a = stk.top();
			stk.pop();
			
			if(a == 2){
				arr[level-1] += arr[level] * 2;
				level--;
			}else{
				check = 0;
				break;
			}
			
		}else{
			if(stk.empty()){
				check = 0;
				break;
			}
			
			int a = stk.top();
			stk.pop();
			
			if(a == 3){
				arr[level-1] += arr[level] * 3;
				level--;
			}else{
				check = 0;
				break;
			}
		}
	}
	
	if(check == 0 || !stk.empty())
		cout << "0";
	else{
		cout << arr[0];
	}
}
