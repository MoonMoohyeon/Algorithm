cycle은 N개의 정점에대해서, N번 이상 정점이 갱신될 떄, 음의 사이클이 존재한다고 알 수 있다
이미 큐에 들어간 정점에 대해선 또 넣지 않는것이 중요하다
정점의 방문 값이 변하는 것만 모아서 처리하는 방법론이다.
음의 가중치가 있어도 가능 << 다익스트라는 음의 가중치가 존재하면 불가능하다

다익스트라 == O(E)
벨만 포드 == O(VE)
SPFA == O(E) ~ O(V+E)이다.



#include <iostream>
#include<vector>
#include<queue>
using namespace std;

const int n_ = 101;
const int INF = 2100000000;

int N, M;
vector<pair<int, int>> v[n_];
int info[n_];
bool gain[n_][n_];
bool table[n_];
int cycle[n_];
int start, goal;

int main() {
	cin >> N >> M;
	for (int i = 0; i < M; i++) {
		int a, b, c;
		cin >> a >> b >> c;

		v[a].push_back({ b,c });
		v[b].push_back({ a,c });
	}

	start = 1;
	goal = N;
	for (int i = 0; i < n; i++) {
		info[i] = INF;
	}
	info[start] = 0;

	queue<pair<int, int>> q;
	q.push({ start , 0 });

	for (;;) {
		int s = q.size();
		
		for (int k = 0; k < s; k++) {
			pair<int, int> prev = q.front();
			q.pop();
			table[prev.first] = 0;
			cycle[prev.first]++;
			
			if(cycle[prev.first] >= N){
				//음의 사이클 존재
			}

			int size = v[prev.first].size();
			for (int i = 0; i < size; i++) {
				int next = v[prev.first][i].first;
				int next_w = info[prev.first] + v[prev.first][i].second;

				if (info[next] > next_w) {
					if(table[next] == 0)
						table[next] = 1;
					else
						continue;
						
					info[next] = next_w;
					q.push({ next, next_w });
				}
			}

		}
	}

	return 0;
}
