#include <iostream>
#include <cstring>
#include <vector>
using namespace std;

const int n_ = 1000001;
string str;
string sen;
int p[n_];

void getPI(){
	string cmp = str;
	p[0] = 0;
	p[1] = 0;
	int l = str.size();
	int idx = 0;
	
	for(int i=2; i<=l;){
		if(str[i-1] == cmp[idx]){
			p[i] = idx + 1;
			idx = idx + 1;
			i++;
		}else{
			//현재 i개 일치. 
			if(idx == 0){
				p[i] = 0;
				i++;
			}
			idx = p[idx];
		}
	}
}

int main() {
	getline(cin, sen);
	getline(cin, str);
	
	getPI();
	for(int i=0; i<=str.size(); i++){
		//cout << p[i] << ' ';
	}
	
	
	vector<int> v;
	int l = sen.size();
	int s = str.size();
	int idx = 0;
	
	
	for(int i=0; i<l;){
		if(sen[i] == str[idx]){
			idx++;
			
			if(idx == s){
				v.push_back(i - idx + 2);
				idx = p[idx];
			}
			
			i++;
		}else{
			if(idx == 0)
				i++;
			idx = p[idx];
		}
	}
	
	cout << v.size() << '\n';
	s = v.size();
	for(int i=0; i<s; i++){
		cout << v[i] << ' ';
	}
	
	
	return 0;
}
