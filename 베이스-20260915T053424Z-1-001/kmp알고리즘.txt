#include <iostream>
#include<cstring>
#include<vector>
using namespace std;

string str;
string fin;
int str_len;
int fin_len;

int pi[1000001];
bool table[1000001];

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
	getline(cin, str);
	getline(cin, fin);
	str_len = str.size();
    fin_len = fin.size();
    
    pi[0] = 0;
    pi[1] = 0;
    int len = 2;
    int ans = 0;
	for(int j = 0; len <= fin_len;){
		if(fin[len-1] == fin[j]){
			pi[len] = ++ans;
			len++;
			j++;
		}
		else{
			if(ans == 0){
				pi[len] = 0;
				len++;
				j = 0;
			}
			j = pi[ans];
			ans = pi[ans];
		}
	}

	
	int ret = 0;
	int pre = 0;
	for(int i=0; i<=str_len - fin_len;){
		if(str[i] == fin[0]){
			int ans = 0;
			ans = pre;
			
			for(int j = pre; j < fin_len; j++){
				if(str[i+j] == fin[j]){
					ans++;
				}else{
					i = i + j - pi[ans];
					pre = pi[ans];
					break;
				}
			}
			
			if(ans == fin_len){
				ret++;
				table[i] = 1;
				i = i + fin_len - pi[ans];
				pre = pi[ans];
			}
		}else{
			i++;
			pre = 0;
		}
	}
	
	cout << ret << '\n';
	for(int i=0; i<str_len; i++){
		if(table[i]){
			cout << i + 1 << ' ';
		}
	}
	
	return 0;
}
