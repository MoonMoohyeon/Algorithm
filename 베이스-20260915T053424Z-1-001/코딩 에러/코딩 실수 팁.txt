			int prev = pq.top();
			pq.pop();
			
			string str = to_string(prev);

				for(int j = 0; j<len; j++){
					if(i != j){
						char c = cmp[i];
						cmp[i] = cmp[j];
						cmp[j] = c;
						int next = stoi(cmp);
							
						if(table[next] == 0){
							table[next] = 1;
							//cout << next << '\n';
							pq.push(next); // ********************************** 이때 만약 push된 값이 우선순위가 높다면, 해당 값이 현재 레벨에서 또 탐색된다.
						}
						
						break;
					}
				}
			}
		}
